cmake_minimum_required(VERSION 3.22.1)
project(netboot_media LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
include("${CMAKE_CURRENT_LIST_DIR}/dependencies.cmake")
include(CheckIncludeFile)
include(CheckSymbolExists)

FetchContent_MakeAvailable(wim udf fat)

set(config "#define PACKAGE_VERSION \"${WIMLIB_VERSION}\"\n#define VERSION \"${WIMLIB_VERSION}\"\n#define HAVE_PTHREAD 1\n#define HAVE_STAT_NANOSECOND_PRECISION 1\n")
foreach(header alloca.h byteswap.h dlfcn.h endian.h errno.h fcntl.h inttypes.h stdarg.h stddef.h stdint.h stdio.h stdlib.h strings.h string.h sys/file.h sys/stat.h sys/syscall.h sys/times.h sys/types.h sys/xattr.h time.h unistd.h utime.h)
    string(TOUPPER "HAVE_${header}" key)
    string(REGEX REPLACE "[/.]" "_" key "${key}")
    check_include_file("${header}" ${key})
    if(${key})
        string(APPEND config "#define ${key} 1\n")
    endif()
endforeach()
foreach(symbol fdopendir flock fsetxattr fstatat futimens lgetxattr llistxattr lsetxattr mempcpy openat posix_fallocate readlinkat utimensat)
    string(TOUPPER "HAVE_${symbol}" key)
    check_symbol_exists(${symbol} "dirent.h;sys/file.h;sys/xattr.h;sys/stat.h;string.h;fcntl.h;unistd.h" ${key})
    if(${key})
        string(APPEND config "#define ${key} 1\n")
    endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "${config}")
file(GLOB wim_sources "${wim_SOURCE_DIR}/src/*.c")
# External-WIM globbing/joining is not part of our API (glob requires Android 28).
list(FILTER wim_sources EXCLUDE REGEX "/(win32_.*|ntfs-3g_.*|wimboot|test_support|reference|join)\\.c$")
add_library(wim STATIC ${wim_sources})
target_include_directories(wim PUBLIC "${wim_SOURCE_DIR}/include" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(wim PRIVATE HAVE_CONFIG_H BUILDING_WIMLIB _GNU_SOURCE _FILE_OFFSET_BITS=64 _LARGEFILE_SOURCE)
target_compile_options(wim PRIVATE -Wno-pointer-sign)
# Avoid unoptimized multi-GB WIM processing in debug builds.
target_compile_options(wim PRIVATE $<$<CONFIG:Debug>:-O2>)

# Discover CDIMAGE's nonzero UDF partition instead of assuming partition zero.
file(READ "${udf_SOURCE_DIR}/src/udfread.c" udf_code)
set(udf_old "_read_vds(input, 0, &vds)")
string(FIND "${udf_code}" "${udf_old}" udf_patch_offset)
if(udf_patch_offset LESS 0)
    message(FATAL_ERROR "Re-evaluate the libudfread partition fix against the pinned source")
endif()
string(REPLACE "${udf_old}" "_read_vds(input, -1, &vds)" udf_code "${udf_code}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/udfread.c" "${udf_code}")
add_library(udf STATIC "${CMAKE_CURRENT_BINARY_DIR}/udfread.c" "${udf_SOURCE_DIR}/src/ecma167.c" "${udf_SOURCE_DIR}/src/default_blockinput.c")
target_include_directories(udf PUBLIC "${udf_SOURCE_DIR}/src")
target_compile_definitions(udf PRIVATE HAVE_UNISTD_H=1 HAVE_FCNTL_H=1 _FILE_OFFSET_BITS=64)

# Apply official FatFs R0.16 patch-1/patch-2 to a build-local copy.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fat")
file(READ "${fat_SOURCE_DIR}/source/ff.c" fat_code)
string(REPLACE "#define MAX_EXFAT" "#define MIN_EXFAT 0x00000100\n#define MIN_FAT12 32\n#define MIN_VOLUME 64\n#define MAX_EXFAT" fat_code "${fat_code}")
string(REPLACE "BPB_TotSec16) >= 128" "BPB_TotSec16) >= MIN_VOLUME" fat_code "${fat_code}")
string(REPLACE "if (nclst <= MAX_FAT12) fmt = FS_FAT12;" "if (nclst <= MAX_FAT12) fmt = FS_FAT12;\n\t\tif (nclst <= MIN_FAT12) fmt = 0;" fat_code "${fat_code}")
string(REPLACE "sz_vol < 128" "sz_vol < MIN_VOLUME" fat_code "${fat_code}")
string(REPLACE "b_data + pau * 16" "b_data + pau * MIN_FAT12" fat_code "${fat_code}")
string(REPLACE "if (ncl > MAX_EXFAT)" "if (ncl < MIN_EXFAT || ncl > MAX_EXFAT)" fat_code "${fat_code}")
string(REPLACE "fs->fsize = fasize;" "if (fasize >= 0x200000) return FR_NO_FILESYSTEM;\n\t\tfs->fsize = fasize;" fat_code "${fat_code}")
string(REPLACE "si < dj.dir[XDIR_NumLabel];" "si < dj.dir[XDIR_NumLabel] && si < 11;" fat_code "${fat_code}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fat/ff.c" "${fat_code}")
foreach(f ff.h diskio.h ffunicode.c)
    configure_file("${fat_SOURCE_DIR}/source/${f}" "${CMAKE_CURRENT_BINARY_DIR}/fat/${f}" COPYONLY)
endforeach()
file(READ "${fat_SOURCE_DIR}/source/ffconf.h" fat_conf)
foreach(pair "FF_USE_MKFS=1" "FF_USE_LFN=2" "FF_LFN_UNICODE=2" "FF_CODE_PAGE=437" "FF_FS_NORTC=1" "FF_LFN_BUF=765")
    string(REPLACE "=" ";" values "${pair}")
    list(GET values 0 name)
    list(GET values 1 value)
    string(REGEX REPLACE "(#define ${name}[ \t]+)[0-9]+" "\\1${value}" fat_conf "${fat_conf}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fat/ffconf.h" "${fat_conf}")
add_library(fat STATIC "${CMAKE_CURRENT_BINARY_DIR}/fat/ff.c" "${CMAKE_CURRENT_BINARY_DIR}/fat/ffunicode.c")
target_include_directories(fat PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/fat")

add_library(netboot_media SHARED media.cpp)
target_compile_definitions(netboot_media PRIVATE _FILE_OFFSET_BITS=64)
target_compile_definitions(netboot_media PRIVATE $<$<CONFIG:Debug>:NETBOOT_MEDIA_TESTING>)
target_compile_options(netboot_media PRIVATE -Wall -Wextra -Werror)
target_link_libraries(netboot_media PRIVATE udf fat wim)
target_link_options(netboot_media PRIVATE -Wl,-z,max-page-size=16384)
